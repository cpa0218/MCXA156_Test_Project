  RCC_OscInitStruct.HSEState = RCC_HSE_ON;
  RCC_OscInitStruct.HSEPredivValue = RCC_HSE_PREDIV_DIV1;
  RCC_OscInitStruct.HSIState = RCC_HSI_ON;

uint16_t ADC_Read(uint32_t Channel)
{
	ADC_ChannelConfTypeDef sConfig = {0};
	sConfig.Channel = Channel;                                         /* 通道 */
	sConfig.Rank = ADC_REGULAR_RANK_1;                              
	sConfig.SamplingTime = ADC_SAMPLETIME_55CYCLES_5;                  /* 采样时间 */
	if (HAL_ADC_ConfigChannel(&hadc1, &sConfig) != HAL_OK)             
	{
		Error_Handler();
	}
	HAL_ADC_Start(&hadc1);
	HAL_ADC_PollForConversion(&hadc1, HAL_MAX_DELAY);
	return (uint16_t)HAL_ADC_GetValue(&hadc1);
}

void delay_us(uint32_t udelay)
{
  uint32_t startval,tickn,delays,wait;
 
  startval = SysTick->VAL;
  tickn = HAL_GetTick();
  //sysc = 72000;  //SystemCoreClock / (1000U / uwTickFreq);
  delays =udelay * 72; //sysc / 1000 * udelay;
  if(delays > startval)
    {
      while(HAL_GetTick() == tickn)
        {
 
        }
      wait = 72000 + startval - delays;
      while(wait < SysTick->VAL)
        {
 
        }
    }
  else
    {
      wait = startval - delays;
      while(wait < SysTick->VAL && HAL_GetTick() == tickn)
        {
 
        }
    }
}

void MQTT_Task() {
	static uint8_t i=0;
	i++;
	if (RxBuffer[0] == (char)0x30) {  // 检测PUBLISH报文
			// 解析主题和消息
			uint16_t topic_len = (RxBuffer[3] << 8) | RxBuffer[4];
			char topic[topic_len + 1];
			memcpy(topic, &RxBuffer[5], topic_len);
			topic[topic_len] = '\0';
			
			uint8_t *msg_start = (uint8_t *)&RxBuffer[5 + topic_len];
			uint16_t msg_len = remaining_length - topic_len - 2;
			printf("Topic: %s, Message: %.*s\n", topic, msg_len, msg_start);
	}
	if(i%50==0)
	{
		// 定义主题和消息
		const char *topic = "topic/stm32";
		char *msg ;
		char pubtext[256];
//		sprintf(msg,"{\"Temp\":%.2f,\"Huim\":%.2f,\"Lux\":%.2f,\"ppm\":%.2f,}",AHT20_Temperature,AHT20_Humidity,Lux,mq2);
		// 构造报文缓冲区
		uint8_t publish_packet[256];
		uint8_t pos = 0;

		// 固定头（类型 + 标志位）
		publish_packet[pos++] = 0x30;  // QoS=0，保留位0

		// 剩余长度（动态计算）
		uint8_t remaining_len = 2 + strlen(topic) + strlen(msg);
		publish_packet[pos++] = remaining_len;

		// 可变头：主题名长度（2字节）
		publish_packet[pos++] = (strlen(topic) >> 8) & 0xFF;  // 高位
		publish_packet[pos++] = strlen(topic) & 0xFF;         // 低位

		// 主题名（ASCII）
		memcpy(&publish_packet[pos], topic, strlen(topic));
		pos += strlen(topic);

		// 有效载荷：消息内容
		memcpy(&publish_packet[pos], msg, strlen(msg));
		pos += strlen(msg);
		
		sprintf(pubtext,"AT+CIPSEND=%d\r\n",pos);//发送连接服务器指令
		esp8266_send_cmd(pubtext,"OK",100);		//发送命令
		// 发送报文
		HAL_UART_Transmit(&huart2, publish_packet, pos, 1000);
// STM32发送前打印报文（调试用）
//		printf("发送的十六进制:");
//		for (int i = 0; i < pos; i++) {
//			printf("%02X ", publish_packet[i]);
//		}		
		memset(pubtext,0,sizeof(pubtext));
	}	
	if(i==100)
	{
	//报文编写
		message[0]=0xC0;              //第1个字节 ：固定0xC0                      
		message[1]=0x00;              //第2个字节 ：固定0x00 
		esp8266_send_cmd("AT+CIPSEND=2\r\n","OK",100);		//发送命令
		HAL_UART_Transmit(&huart2, message, 2, 1000);	
	//清空报文	
		memset(message,0,2);
		i=0;
	}
}

	}
	else

		RxBuffer[Uart2_Rx_Cnt] = aRxBuffer;   //接收数据转存
		if(receive_flag==0)

		if(RxBuffer[Uart2_Rx_Cnt-Uart2_count]=='<')
		{	
			Uart2_count++;
		if((RxBuffer[Uart2_Rx_Cnt]=='>')||Uart2_count>=14)
		{
			uint8_t My_i=0;
		for(int i=Uart2_Rx_Cnt-Uart2_count+1;i<Uart2_Rx_Cnt+1;i++)
			my_order[My_i++]=RxBuffer[i];
			receive_flag=1;
			Uart2_count=0;
		}
	}
	}
		Uart2_Rx_Cnt++;
	
	}
	
	HAL_UART_Receive_IT(&huart2, (uint8_t *)&aRxBuffer, 1);   //再开启接收中断
}
